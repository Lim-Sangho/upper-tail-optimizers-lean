\subsection{The local reduction}\label{sec:local-reduction}

Let $H$ be a $d$-regular graph with $d\ge2$ and write $m=|E(H)|$, so that
the homomorphism density constraint reads $t(H,W)\ge r^m$.
We obtain the reduction uniformly in a sufficiently small neighborhood of
each fixed $r_0\in(0,1)\setminus\{r_*\}$.
\Cref{lem:active-constraint,lem:edge-density-deficit} show that optimizers
saturate the $H$-density constraint and have edge density below $r$ on the
symmetry-breaking side.  The proof of boundary convergence in
\Cref{lem:boundary-convergence} is deferred to \Cref{app:local-reduction}.

\begin{lemma}[Active constraint]\label{lem:active-constraint}
Let $0<p<r<1$.  Every optimizer of \eqref{eq:graphon-variational-problem} satisfies
$t(H,W)=r^m$.
\end{lemma}

\begin{proof}
Let $W$ be an optimizer, and suppose for contradiction that $t(H,W)>r^m$.
For $0\le\theta\le1$, define
\[
  W_\theta=p+\theta(W-p).
\]
Then, $W_0\equiv p$ has $t(H,W_0)=p^m<r^m$, while $W_1=W$ has
$t(H,W)>r^m$.  By continuity of $t(H,\cdot)$, there exists
$0<\theta_0<1$ with $t(H,W_{\theta_0})=r^m$.  Since $J_p$ is strictly convex
and uniquely minimized at $p$ with $J_p(p) = 0$, we have, for every
$(x,y)\in[0,1]^2$,
\begin{align*}
  J_p(p+\theta_0(W(x,y)-p))
  & {}
  \le
  (1-\theta_0)J_p(p)+\theta_0J_p(W(x,y))
  =
  \theta_0J_p(W(x,y))
  {}
  \le
  J_p(W(x,y)),
\end{align*}
with strict inequality unless $W(x,y)=p$.  Integrating gives
\[
   I_p(W_{\theta_0}) \le I_p(W).
\]
On the other hand, $W_{\theta_0}$ is feasible, so the optimality of $W$ gives
$I_p(W_{\theta_0})\ge I_p(W)$.  Hence, $I_p(W_{\theta_0})=I_p(W)$.
Since the pointwise difference is nonnegative, the equality of the integrals forces
\[
  J_p(p+\theta_0(W(x,y)-p))=J_p(W(x,y))
\]
for almost every $(x,y)$, and therefore $W(x,y)=p$ almost everywhere.  This
would imply $t(H,W)=p^m<r^m$, contradicting $t(H,W)>r^m$.  Thus,  every
optimizer satisfies $t(H,W)=r^m$.
\end{proof}

\begin{lemma}[Strict edge-density deficit]\label{lem:edge-density-deficit}
Let $r\in(0,1)\setminus\{r_*\}$ and $0<p<\pc(r)$.
If $W$ is an optimizer of \eqref{eq:graphon-variational-problem}, then $e(W)<r$.
\end{lemma}

\begin{proof}
By condition (M1) of \Cref{thm:lz-boundary}, for $p<\pc(r)$, the point $(r^d,J_p(r))$ is not on the convex minorant of
$x\mapsto J_p(x^{1/d})$. \Cref{thm:lz-criterion} therefore shows that $W\equiv r$ is not a
minimizer. Since $W\equiv r$ is feasible, every optimizer satisfies $I_p(W)<J_p(r)$.
By Jensen's inequality,
$
  I_p(W)\ge J_p(e(W)).
$
Since $p<\pc(r)<r$, the function $J_p$ is strictly increasing on $[r,1]$.
If $e(W)\ge r$, it is a contradiction since
\[
  I_p(W)\ge J_p(e(W))\ge J_p(r).
  \qedhere
\]
\end{proof}

\begin{lemma}[Boundary convergence of optimizers]\label{lem:boundary-convergence}
Let $r_n\in(0,1)\setminus\{r_*\}$ and $p_n\in(0,\pc(r_n))$ satisfy
\[
  r_n\to r_0\in(0,1)\setminus\{r_*\},
  \qquad
  \pc(r_n)-p_n\to0.
\]
For each $n$, let $W_n\in\W_H(p_n,r_n)$.  Then, writing $r_0$ also for the constant
graphon with value $r_0$,
\[
  \delta_\square(W_n,r_0)\to0,
  \qquad
  e(W_n)\to r_0.
\]
\end{lemma}

For $0\le\eps\le1$ and $0\le\tau\le1$, define the fixed-$(e,t(H,\cdot))$ \emph{entropy
envelope}
\begin{equation}\label{eq:fixed-density-entropy}
  S(\eps,\tau)=
  \sup\{s(W):e(W)=\eps,\ t(H,W)=\tau\},
\end{equation}
with the convention $\sup\varnothing=-\infty$.
When the maximum is attained by a bipodal graphon and
the maximizer is unique up to relabeling, denote it by $B_{\eps,\tau}$.
For $p,r\in(0,1)$ and $0\le\eps\le1$, define the \emph{reduced objective}
\begin{equation}\label{eq:reduced-objective}
  I_{p,r}(\eps)=
  \inf\{I_p(W):e(W)=\eps,\ t(H,W)=r^m\},
\end{equation}
with the convention $\inf\varnothing=+\infty$.
Taking the infimum in \eqref{eq:relative-entropy-identity} gives
\[
  I_{p,r}(\eps)
  =-2S(\eps,r^m)-\log(1-p)+\eps\log\frac{1-p}{p}.
\]
A graphon attains this infimum exactly when it maximizes entropy under
the same constraints.

For $\eps<r$, define the \emph{edge-density deficit}
\[
  \delta:=r-\eps.
\]
By \Cref{lem:edge-density-deficit},
this deficit is positive for every optimizer on the symmetry-breaking side.
Every optimizer of the original problem satisfies $t(H,W)=r^m$ by
\Cref{lem:active-constraint}.  We can therefore first minimize the cost at
each fixed deficit $\delta$, then minimize over $\delta$.
Near the phase boundary on the symmetry-breaking side, the next lemma
places every optimizer's deficit in $(0,\delta_0)$ for some $\delta_0>0$,
uniformly for $r$ near $r_0$.  For each $\delta$ in this interval,
\Cref{thm:krrs-analytic-extension} identifies the entropy maximizer at
densities $(r-\delta,r^m)$ as a unique bipodal graphon $B_{r-\delta,r^m}$,
up to relabeling.
Thus, only the choice of $\delta$ remains: \Cref{cor:one-dimensional-reduction} shows
that minimizing $I_{p,r}(r-\delta)$ over $0<\delta<\delta_0$ is equivalent to the
original problem, with a one-to-one correspondence between minimizing deficits
and graphon optimizers modulo relabeling.
Its proof is deferred to \Cref{app:local-reduction}.

\begin{lemma}[Uniform localization and fixed-density bipodality]\label{lem:fixed-density-bipodality}
Fix $r_0\in(0,1)\setminus\{r_*\}$.
There exist an open interval $I$ containing $r_0$, with compact closure
$\bar I \subseteq (0,1)\setminus\{r_*\}$, and a constant $\delta_0>0$ such that,
for every $r\in I$ and $0<\delta<\delta_0$, the entropy maximizer with edge density
$r-\delta$ and $H$-density $r^m$ is bipodal and unique up to relabeling.
Thus $B_{r-\delta,r^m}$ is well defined up to relabeling, and for every $p\in(0,1)$,
\begin{equation}\label{eq:reduced-objective-attainment}
  S(r-\delta,r^m)=s(B_{r-\delta,r^m}),
  \qquad
  I_{p,r}(r-\delta)=I_p(B_{r-\delta,r^m}).
\end{equation}
Furthermore, there exists $0<\eta<\inf_{r\in I}\pc(r)$ such that,
whenever $r\in I$ and $\pc(r)-\eta<p<\pc(r)$, every optimizer
$W$ of \eqref{eq:graphon-variational-problem} satisfies
\[
  0<r-e(W)<\delta_0.
\]
\end{lemma}

\begin{proof}
Apply \Cref{thm:krrs-analytic-extension} at $\eps_0=r_0$, obtaining
an open neighborhood $U \subseteq (0,1)\setminus\{r_*\}$ of $r_0$ and a width $\Delta>0$
for the extended analytic parameter map over
$(\eps, \vartheta) \in U \times (0,\Delta)$.
Choose an open interval $I$ containing $r_0$ such that $\bar I \subseteq U$.
Choose $\delta_0>0$ so small that
$[r-\delta_0,r]\subseteq U$ for every $r\in\bar I$ and
$m\delta_0<\Delta$.
For $r\in I$ and $0<\delta<\delta_0$, the mean value theorem gives
\[
  0<r^m-(r-\delta)^m\le m\delta<\Delta.
\]
Thus, \Cref{thm:krrs-analytic-extension} with
$(\eps,\vartheta)=(r-\delta,r^m-(r-\delta)^m) \in U \times (0,\Delta)$
supplies $B_{r-\delta,r^m}$ that satisfies
$e(B_{r-\delta,r^m})=r-\delta$ and $t(H,B_{r-\delta,r^m})=r^m$.
It maximizes entropy among graphons with these densities,
and is unique up to relabeling.  The definition of $S$ then gives
\eqref{eq:reduced-objective-attainment} for every $p\in(0,1)$.

We next claim that, for some
$0<\eta<\inf_{r\in I}\pc(r)$, every optimizer
$W$ of \eqref{eq:graphon-variational-problem} with
$r\in I$ and $\pc(r)-\eta<p<\pc(r)$ satisfies
\begin{equation}\label{eq:edge-density-localization}
  0<r-e(W)<\delta_0.
\end{equation}
Otherwise, taking $\eta_n\to0$, there
are $r_n\in I$, $0<p_n<\pc(r_n)$ with $\pc(r_n)-p_n\to0$, and optimizers
$W_n$ violating \eqref{eq:edge-density-localization}.
By \Cref{lem:edge-density-deficit}, this means $e(W_n)\le r_n-\delta_0$.
Passing to a subsequence, compactness of $\bar I$ gives
$r_n\to\bar r$ for some $\bar r\in\bar I$.
Then \Cref{lem:boundary-convergence} gives $e(W_n)\to\bar r$,
contradicting $e(W_n)\le r_n-\delta_0$.
\end{proof}

\begin{corollary}[Uniform one-dimensional reduction]\label{cor:one-dimensional-reduction}
Let $I$, $\delta_0$, and $\eta$ be as in \Cref{lem:fixed-density-bipodality}.
For every $r\in I$ and $p\in(\pc(r)-\eta,\pc(r))$, the variational problem
defining $\Phi_H(p,r)$ in \eqref{eq:graphon-variational-problem} reduces to the
one-dimensional problem of choosing the deficit $\delta$:
\begin{equation}\label{eq:edge-density-minimization}
  \Phi_H(p,r)=
  \min_{0<\delta<\delta_0}I_{p,r}(r-\delta).
\end{equation}
Furthermore, for such $p,r$, any optimizer $\widehat W$ attaining $\Phi_H(p,r)$
is $B_{r-\widehat\delta,r^m}$ up to relabeling, where $\widehat\delta:=r-e(\widehat W)$
attains the minimum in the right-hand side of \eqref{eq:edge-density-minimization}.
Conversely, any such minimizer $\widehat\delta$ gives an optimizer $B_{r-\widehat\delta,r^m}$ of
\eqref{eq:graphon-variational-problem}.  Consequently, $\widehat W\mapsto\widehat\delta$ is
a bijection between graphon optimizers modulo relabeling and minimizers of
\eqref{eq:edge-density-minimization}.
\end{corollary}
