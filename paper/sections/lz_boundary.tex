\section{The Lubetzky--Zhao boundary}\label{sec:lz-boundary}
Fix $d\ge2$, and write
\[
  r_*:=\frac{d-1}{d},
  \qquad
  p_*:=\frac{d-1}{(d-1)+\exp(d/(d-1))}.
\]
We use $z$ for the density and $x=z^d$ for the coordinate in the
Lubetzky--Zhao criterion.  For $p\in(0,1)$, define
\[
  \varphi_{p,d}(x)=J_p(x^{1/d})\qquad (0\le x\le1).
\]
Thus, a contact at coordinate $x=r^d$ corresponds to density $z=r$;
supporting lines below are affine in $x$.

For a continuous function $f$ on a compact interval $I$, the \emph{convex minorant}
of $f$ is the largest convex function $g:I\to\mathbb R$ satisfying
$g(x)\le f(x)$ for every $x\in I$.
Recall that the result of Lubetzky and Zhao~\cite{lubetzky2015large} states
that when $H$ is $d$-regular, the constant graphon $W\equiv r$ is the unique minimizer
of~\eqref{eq:graphon-variational-problem} if and only if
$(r^d, J_p(r))$ lies on the convex minorant of $\varphi_{p,d}$.
We analyze $\varphi_{p,d}$ and its convex minorant for
\Cref{thm:nonexceptional-optimizers,thm:endpoint-optimizers}.
Away from $r_*$, the density gap has a positive quadratic leading term at
each of the two contacts.  At $r_*$, they merge into a single contact with
a positive quartic leading term, as the following theorem records.
\begin{theorem}[Lubetzky--Zhao boundary]\label{thm:lz-boundary}
There exist real-analytic functions
\[
  \pc,\sc:(0,1)\to(0,1),
\]
with $\pc(r_*)=p_*$ and
$\sc(r_*)=r_*$, satisfying the following.
\begin{enumerate}[label=(M\arabic*)]
  \item 
  For $p,r\in(0,1)$, the point $(r^d,J_p(r))$ lies on the convex minorant
  of $\varphi_{p,d}$ if and only if $p\ge\pc(r)$.
  Thus, $\pc$ defines the \emph{Lubetzky--Zhao boundary}, with
  $0<\pc(r)<\min\{r,p_*\}$ for $r\ne r_*$.

  \item
  Let $\ell_r$ be the tangent line of $\varphi_{\pc(r),d}$ at $x=r^d$:
  \[
    \ell_r(x)=\varphi_{\pc(r),d}(r^d)
      +\varphi_{\pc(r),d}'(r^d)(x-r^d).
  \]
  It satisfies $\ell_r(x)\le\varphi_{\pc(r),d}(x)$ for all $x\in[0,1]$,
  with equality exactly when $x\in\{r^d,\sc(r)^d\}$.
  The lines vary continuously with $r$; the contacts are distinct for
  $r\ne r_*$ and merge at $x=r_*^d$ as $r\to r_*$.

  \item
  For $r\ne r_*$, define the supporting cost gap in the density variable by
  \begin{equation}\label{eq:supporting-cost-gap}
    g_{d,r}(z):=J_{\pc(r)}(z)-\ell_r(z^d)
    \qquad(0\le z\le1).
  \end{equation}
  For every compact $K\subseteq(0,1)\setminus\{r_*\}$, uniformly for
  $r\in K$ and $u\in\{r,\sc(r)\}$,
  \begin{equation}\label{eq:contact-gap-expansion}
    g_{d,r}(z)=\frac12g_{d,r}''(u)(z-u)^2+O_{d,K}(|z-u|^3)
    \qquad(z\to u),
  \end{equation}
  where $0<c_{d,K}\le g_{d,r}''(u)\le C_{d,K}<\infty$.

  \item
  At $r=r_*$, define the supporting cost gap in the density variable by
  \begin{equation}\label{eq:endpoint-supporting-gap}
    \Gamma_d(z):=J_{p_*}(z)-\ell_{r_*}(z^d)
    \qquad(0\le z\le1).
  \end{equation}
  It is the limit of $g_{d,r}(z)$ as $r\to r_*$ and satisfies
  \begin{equation}\label{eq:endpoint-gap-expansion}
    \Gamma_d(z)=\frac{d^5}{24(d-1)^2}(z-r_*)^4
      +O_d(|z-r_*|^5)
    \qquad(z\to r_*).
  \end{equation}
\end{enumerate}
\end{theorem}

The convexity threshold and the common-tangent description used
below already appear in the analysis of Lubetzky and Zhao~\cite[Appendix~A]{lubetzky2015large}.
Here, we record the details for the later argument.  In particular, we establish
analyticity of $\pc$ and $\sc$ through $r_*$, coalescence of the contacts, and the
quadratic and quartic expansions of the gap in the density variable.
The proofs of the following two lemmas are in
Appendix~\ref{app:lz-geometry}.

We first locate the convex and concave regions of $\varphi_{p,d}$.
The next lemma shows that $\varphi_{p,d}''(z^d)$ has the same sign as
$h_{p,d}(z)$, defined for $p\in(0,1)$ by
\[
  h_{p,d}(z):=zJ_p''(z)-(d-1)J_p'(z)
  \qquad (0<z<1).
\]

\begin{lemma}[Convexity defect]\label{lem:convexity-defect}
For every $p\in(0,1)$,
\[
  h_{p,d}'(z)=\frac{d(z-r_*)}{z(1-z)^2}
  \qquad\text{and}\qquad
  h_{p,d}(r_*)=d-(d-1)\log\frac{(d-1)(1-p)}{p}.
\]
Thus, $h_{p,d}$ is strictly decreasing on $(0,r_*)$, strictly increasing on
$(r_*,1)$, and has its unique minimum at $r_*$.  Moreover, for
$x\in(0,1)$ and $z=x^{1/d}$,
\[
  \varphi_{p,d}''(x)=\frac{h_{p,d}(z)}{d^2z^{2d-1}},
\]
so $\varphi_{p,d}''(x)$ and $h_{p,d}(z)$ have the same sign.  If $p\ge p_*$,
then $h_{p,d}\ge0$ and $\varphi_{p,d}$ is convex on $[0,1]$.  If $p<p_*$,
then $h_{p,d}$ has exactly two zeros $z_-(p)$ and $z_+(p)$ with $0<z_-(p)<r_*<z_+(p)<1$,
is positive on $(0,z_-(p))\cup(z_+(p),1)$, and is negative on
$(z_-(p),z_+(p))$.  Accordingly, $\varphi_{p,d}$ is strictly convex on
$(0,z_-(p)^d)$ and $(z_+(p)^d,1)$ and strictly concave on
$(z_-(p)^d,z_+(p)^d)$.
\end{lemma}

When $p<p_*$, the graph of $\varphi_{p,d}$ has one concave region between two strictly convex regions.
The convex minorant of $\varphi_{p,d}$ consists of the two convex pieces and an affine face that covers the concave region.  
Moreover, at the two endpoints of this affine face, the line must be tangent to the graph, so that it joins smoothly onto the remaining convex pieces and stays below the graph everywhere.
The following lemma proves this formally. 

\begin{lemma}[Contact points]\label{lem:contact-points}
For every $p\in(0,p_*)$, there are two \emph{contact points} $x_a(p)$ and $x_b(p)$ such that
\[
  0<x_a(p)<z_-(p)^d<r_*^d<z_+(p)^d<x_b(p)<1,
\]
where $z_-(p)$ and $z_+(p)$ are the two zeros of $h_{p,d}$, and
\begin{equation}\label{eq:contact-identity}
  \varphi_{p,d}'(x_a(p))=\varphi_{p,d}'(x_b(p)),
  \qquad
  \varphi_{p,d}(x_b(p))-\varphi_{p,d}(x_a(p))
  =\varphi_{p,d}'(x_a(p))(x_b(p)-x_a(p)).
\end{equation}
The convex minorant of $\varphi_{p,d}$ agrees with $\varphi_{p,d}$ outside
$[x_a(p),x_b(p)]$; on $[x_a(p),x_b(p)]$, it is the straight line segment joining
the two contact points, and it lies strictly below $\varphi_{p,d}$ on the open
interval $(x_a(p),x_b(p))$.  The maps $p\mapsto x_a(p),x_b(p)$ are analytic on
$(0,p_*)$, with $dx_a/dp>0$ and $dx_b/dp<0$.
As $p\uparrow p_*$, both $x_a(p)$ and $x_b(p)$ converge to $r_*^d$;
as $p\downarrow0$, $x_a(p)\downarrow0$ and $x_b(p)\uparrow1$.
\end{lemma}

We finally prove Theorem~\ref{thm:lz-boundary}.

\begin{proof}[Proof of Theorem~\ref{thm:lz-boundary}]
\emph{Construction and (M1)--(M2).}
Let $z_a(p):=x_a(p)^{1/d}$ and $z_b(p):=x_b(p)^{1/d}$ be the contact densities.
By \Cref{lem:contact-points}, these are respectively increasing and decreasing
bijections from $(0,p_*)$ onto $(0,r_*)$ and $(r_*,1)$.
For $r<r_*$, define $\pc(r)=z_a^{-1}(r)$ and $\sc(r)=z_b(\pc(r))$;
for $r>r_*$, define $\pc(r)=z_b^{-1}(r)$ and $\sc(r)=z_a(\pc(r))$.
In particular, $0<\pc(r)<p_*$ for $r\ne r_*$.
The limits give continuous extensions to $r \in (0,1)$ with
$\pc(r_*)=p_*$ and $\sc(r_*)=r_*$.

For $p<p_*$, \Cref{lem:contact-points} says that $(r^d,J_p(r))$ lies on the
convex minorant exactly when
\[
  r\notin(z_a(p),z_b(p))
  \quad\Longleftrightarrow\quad p\ge\pc(r),
\]
by monotonicity and the definition of $\pc$; at $r=r_*$, both conditions
are false.  For $p\ge p_*$, the curve is convex by
\Cref{lem:convexity-defect} and $p\ge\pc(r)$ automatically.
For (M1), it remains to prove $\pc(r)<r$ for $r\ne r_*$.
Since $z_b(p)>r_*>p_*>\pc(r)$, the contact identity gives
\[
  \varphi_{\pc(r),d}'(x_a(\pc(r)))=\varphi_{\pc(r),d}'(x_b(\pc(r)))>0.
\]
Since $\varphi_{p,d}'(z^d)=J_p'(z)/(d z^{d-1})$ has the sign of $z-p$,
we have $\pc(r)<z_a(\pc(r))\le r$.

For (M2), when $r\ne r_*$, \Cref{lem:contact-points} identifies $\ell_r$
with the affine part of the convex minorant between $r^d$ and $\sc(r)^d$.
Its extension supports the minorant on $[0,1]$; strict separation between
the contacts and strict convexity outside give equality with
$\varphi_{\pc(r),d}$ only at the two contacts.
Continuity of $\pc$ and $\sc$ gives continuity of $\ell_r$ and coalescence
of the contacts at $r_*^d$.  At $p=p_*$, \Cref{lem:convexity-defect} gives
$\varphi_{p_*,d}''(x)>0$ except at $r_*^d$, so $\varphi_{p_*,d}$ is strictly
convex and its tangent there has no other contact.  This proves (M2), and hence
\begin{equation}\label{eq:endpoint-gap-positivity}
  \Gamma_d(z)\ge0,
  \qquad
  \Gamma_d(z)=0\iff z=r_*.
\end{equation}
Continuity also gives $\Gamma_d(z)=\lim_{r\to r_*}g_{d,r}(z)$.

\emph{Analyticity.}
By \Cref{lem:contact-points}, $z_a$ and $z_b$ are analytic with nonzero
derivatives.  The analytic inverse function theorem therefore gives
analyticity of $\pc$ and $\sc$ away from $r_*$.
To treat $r_*$, we normalize the contact equations so that they extend to
coincident contacts.  For $a,b,p\in(0,1)$ with $a\ne b$, set
\begin{align*}
  F_1(a,b,p)&:=\frac{\varphi_{p,d}'(b)-\varphi_{p,d}'(a)}{b-a},\\
  F_2(a,b,p)&:=\frac{\varphi_{p,d}(b)-\varphi_{p,d}(a)
    -\frac12(b-a)\{\varphi_{p,d}'(a)+\varphi_{p,d}'(b)\}}{(b-a)^3}.
\end{align*}
Here $a,b$ are coordinates of the form $x=z^d$; for $a=x_a(p)$ and $b=x_b(p)$,
$F_1=F_2=0$ is equivalent to \eqref{eq:contact-identity}.  Taylor expansion gives
symmetric, jointly analytic extensions across $a=b$, with
\[
  F_1(a,a,p)=\varphi_{p,d}''(a),
  \qquad
  F_2(a,a,p)=-\frac1{12}\varphi_{p,d}'''(a).
\]
By \Cref{lem:convexity-defect}, $h_{p_*,d}(r_*)=h_{p_*,d}'(r_*)=0$ and
$h_{p_*,d}''(r_*)=d^4/(d-1)$.  Differentiating the formula for
$\varphi_{p,d}''$ in that lemma gives
\begin{equation}\label{eq:endpoint-contact-derivatives}
  \varphi_{p_*,d}''(r_*^d)=\varphi_{p_*,d}'''(r_*^d)=0,
  \qquad
  (d r_*^{d-1})^4\varphi_{p_*,d}^{(4)}(r_*^d)
  =\frac{d^5}{(d-1)^2}>0.
\end{equation}
Also, $\partial_p h_{p,d}(r_*)=(d-1)/\{p(1-p)\}>0$, so
$\partial_p\varphi_{p,d}''(r_*^d)>0$.
By symmetry, each $b$-derivative at $a=b$ is half the derivative of
$F_i(a,a,p)$ with respect to $a$.  Hence
\[
  \left.\det D_{(b,p)}(F_1,F_2)\right|_{(a,b,p)=(r_*^d,r_*^d,p_*)}
  =\frac{\varphi_{p_*,d}^{(4)}(r_*^d)}{24}
    \left.\partial_p\varphi_{p,d}''(r_*^d)\right|_{p=p_*}>0.
\]
The analytic implicit function theorem now gives locally unique analytic
solutions $(b,p)$ as functions of $a$ near $r_*^d$.
The continuous pair $(\sc(r)^d,\pc(r))$ solves $F_1=F_2=0$ at $a=r^d$
and equals $(r_*^d,p_*)$ at $r=r_*$, so local uniqueness identifies it
with these solutions.  Composing with $a=r^d$ and taking the positive
$d$th root of $b$ prove analyticity at $r_*$.

\emph{Proof of (M3).}
Fix a compact $K\subseteq(0,1)\setminus\{r_*\}$.
For $r\in K$ and either contact density $u\in\{r,\sc(r)\}$, tangency gives
$g_{d,r}(u)=g_{d,r}'(u)=0$.  Moreover, \Cref{lem:convexity-defect,lem:contact-points} give
\[
  g_{d,r}''(u)
  =J_{\pc(r)}''(u)-\frac{d-1}{u}J_{\pc(r)}'(u)
  =\frac{h_{\pc(r),d}(u)}{u}>0.
\]
The contact densities vary continuously with $r$ and stay uniformly away
from $0$ and $1$.  Thus, continuity and compactness give the stated bounds
on $g_{d,r}''(u)$.  Since $g_{d,r}(z)$ is jointly analytic near the contacts, its
third $z$-derivative is uniformly bounded in neighborhoods of a common radius.
Then, Taylor's theorem gives \eqref{eq:contact-gap-expansion}.

\emph{Proof of (M4).}
Since $\Gamma_d(z)=\varphi_{p_*,d}(z^d)-\ell_{r_*}(z^d)$,
tangency and \eqref{eq:endpoint-contact-derivatives} give, by the chain rule,
$\Gamma_d^{(j)}(r_*)=0$ for $0\le j\le3$ and
$\Gamma_d^{(4)}(r_*)=d^5/(d-1)^2$.
Then, Taylor's theorem gives \eqref{eq:endpoint-gap-expansion}.
\end{proof}

The local expansions in (M3)--(M4) also yield global lower bounds on $[0,1]$.
Fix a compact $K\subseteq(0,1)\setminus\{r_*\}$.
The contacts are uniformly separated on $K$, so \eqref{eq:contact-gap-expansion}
extends the ratio
\[
  \frac{g_{d,r}(z)}{\dist(z,\{r,\sc(r)\})^2}
\]
continuously to $(r,z) \in K\times[0,1]$, with value $g_{d,r}''(u)/2>0$ at each contact
$u\in\{r,\sc(r)\}$.  It is positive elsewhere by (M2), so compactness gives
a positive minimum $\gamma_{d,K}$ and hence
\begin{equation}\label{eq:contact-quadratic-separation}
  g_{d,r}(z)\ge\gamma_{d,K}\,\dist(z,\{r,\sc(r)\})^2
  \qquad(r\in K,\ 0\le z\le1).
\end{equation}

Similarly, \eqref{eq:endpoint-gap-expansion} extends the ratio
$\Gamma_d(z)/|z-r_*|^4$ continuously to $z\in[0,1]$, with value
$d^5/[24(d-1)^2]>0$ at $r_*$.
It is positive elsewhere by \eqref{eq:endpoint-gap-positivity}, so its
minimum $c_d>0$ gives
\begin{equation}\label{eq:endpoint-gap-quartic-bound}
  \Gamma_d(z)\ge c_d|z-r_*|^4
  \qquad(0\le z\le1).
\end{equation}
The quadratic bound \eqref{eq:contact-quadratic-separation} controls
the cost of an edge-density deficit in \Cref{sec:nonexceptional-quadratic-growth};
the quartic bound \eqref{eq:endpoint-gap-quartic-bound} provides the
$L^4$-localization in \Cref{sec:localization-rank-one}.
