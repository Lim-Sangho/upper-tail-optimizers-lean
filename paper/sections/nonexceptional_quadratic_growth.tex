\subsection{The quadratic cost of an edge-density deficit}\label{sec:nonexceptional-quadratic-growth}

Fix $r_0\in(0,1)\setminus\{r_*\}$ and let $I$, $\delta_0$, and $\eta$ be as in
\Cref{lem:fixed-density-bipodality}.  All estimates below are uniform for
$r\in I$.  We may shrink $I$ around $r_0$ and decrease $\delta_0$, decreasing
$\eta$ as needed to retain the conclusions of \Cref{cor:one-dimensional-reduction}.
At the phase boundary $p=\pc(r)$, define the \emph{boundary excess}
\begin{equation}\label{eq:boundary-excess}
  G_r(\delta)=I_{\pc(r),r}(r-\delta)-J_{\pc(r)}(r)
  \qquad (0<\delta<\delta_0).
\end{equation}
This quantity is finite by \Cref{lem:fixed-density-bipodality} and measures
the cost of lowering the edge density from $r$ to $r-\delta$
while keeping the $H$-density at $r^m$.  We will prove $G_r(\delta)$ is quadratic in $\delta$
with a positive leading coefficient:
\[
  G_r(\delta)=\frac12A_H(r)\delta^2+O_{H,I}(\delta^3)
  \qquad (\delta\downarrow0),
  \qquad A_H(r)>0.
\]

Recall the supporting cost gap $g_{d,r}$ from \eqref{eq:supporting-cost-gap},
expressed in the density variable $z$.  It vanishes exactly at $r$ and
$\sc(r)$, and \eqref{eq:contact-quadratic-separation} bounds it
below by a positive multiple of the squared distance to these two densities.
A direct calculation using $\pc(r)<r$ shows that the line $\ell_r$ has positive slope.
For a graphon $W$ with $t(H,W)\ge r^m$, the generalized H\"older inequality
\eqref{eq:generalized-holder} gives $\int W^d\ge r^d$, so
\[
  I_{\pc(r)}(W)-J_{\pc(r)}(r)
  =\int g_{d,r}(W)+\ell_r\!\left(\int W^d\right)-\ell_r(r^d)
  \ge\int g_{d,r}(W).
\]
Thus, to prove the quadratic lower bound in the following lemma, it suffices
to establish $\int g_{d,r}(W)\ge C_{d,I}\delta^2$.

\begin{lemma}[Quadratic lower bound]\label{lem:quadratic-lower-bound}
Let $I$ and $\delta_0$ be as in \Cref{lem:fixed-density-bipodality}.
There exists a constant $C_{d,I}>0$ such that, for every $r\in I$ and
$0<\delta<\delta_0$, any graphon $W$ with $e(W)=r-\delta$ and $t(H,W)\ge r^m$ satisfies
\begin{equation}\label{eq:quadratic-lower-bound}
  I_{\pc(r)}(W)\ge J_{\pc(r)}(r)+C_{d,I}\delta^2.
\end{equation}
\end{lemma}

\begin{proof}
As above, it suffices to establish a quadratic lower bound for $\int g_{d,r}(W)$.
Set $s:=\sc(r)$ and define
\[
  k:=\frac{s^d-r^d}{s-r}>0,
  \qquad
  R_r(z):=z^d-r^d-k(z-r).
\]
The choice of $k$ makes $R_r(r)=R_r(s)=0$, so $R_r$ vanishes at the same
two densities as $g_{d,r}$.  Its integral can be bounded directly using the
two constraints:
\[
  \int R_r(W)=\int W^d-r^d+k\delta\ge k\delta.
\]
We now bound $g_{d,r}(z)$ below by a uniform positive multiple of $R_r(z)^2$.
Cauchy--Schwarz will then give a lower bound on $\int g_{d,r}(W)$ in terms of
$\left(\int R_r(W)\right)^2$, which is at least $k^2\delta^2$.

First, we estimate $|R_r(z)|$ using its zeros and derivative.
Since $0<r,s<1$, we have 
\[
  k=s^{d-1}+s^{d-2}r+\cdots+r^{d-1}\le d.
\]
Hence $|R_r'(z)|=|d z^{d-1}-k|\le d$ on $[0,1]$, so $R_r$ is
$d$-Lipschitz.  For each $z\in[0,1]$, choose $u\in\{r,s\}$ nearest to $z$, so that
$|z-u|=\dist(z,\{r,s\})$.  Since $R_r(u)=0$,
\[
  |R_r(z)|=|R_r(z)-R_r(u)|\le d|z-u|
  =d\,\dist(z,\{r,s\}).
\]
Applying \eqref{eq:contact-quadratic-separation} on $\bar I$ gives
\begin{equation*}
  g_{d,r}(z)
  \ge\gamma_{d,\bar I}\dist(z,\{r,s\})^2
  \ge\frac{\gamma_{d,\bar I}}{d^2}R_r(z)^2.
\end{equation*}
By continuity and compactness of $\bar I$, choose $\eta_{d,I}>0$ such that
$r,s\ge\eta_{d,I}$ uniformly for $r\in I$, so $k\ge\eta_{d,I}^{d-1}$.
Integrating the preceding bound and applying Cauchy--Schwarz, we obtain
\begin{equation*}
  \int g_{d,r}(W)
  \ge\frac{\gamma_{d,\bar I}}{d^2}\int R_r(W)^2
  \ge\frac{\gamma_{d,\bar I}}{d^2}\left(\int R_r(W)\right)^2
  \ge C_{d,I}\delta^2,
\end{equation*}
where $C_{d,I}:=\gamma_{d,\bar I}\eta_{d,I}^{2(d-1)}/d^2>0$.
The last inequality uses $\int R_r(W)\ge k\delta$ and the lower bound on $k$.
\end{proof}

We now prove that the preceding quadratic lower bound is tight.
For small $\delta>0$, we construct a bipodal graphon with
edge density $r-\delta$, $H$-density $r^m$, and $I_{\pc(r)}$-value only
$O_{H,I}(\delta^2)$ above that of the constant graphon.  The construction starts
from the constant graphon $r$, makes a small block, uses the second contact density
$\sc(r)$ on the cross edges, and then adjusts the large-block density to
restore the two constraints.

\begin{lemma}[Quadratic upper bound]\label{lem:quadratic-upper-bound}
  Let $I$ and $\delta_0$ be as in \Cref{lem:fixed-density-bipodality}.
There exists $C_{H,I}>0$ such that, for $r\in I$ and $0<\delta<\delta_0$,
there is a bipodal graphon $W_{r,\delta}$ satisfying $e(W_{r,\delta})=r-\delta$,
$t(H,W_{r,\delta})=r^m$, and
\begin{equation}\label{eq:quadratic-upper-bound}
  I_{\pc(r)}(W_{r,\delta})
  \le
  J_{\pc(r)}(r)+C_{H,I}\delta^2.
\end{equation}
\end{lemma}

\begin{proof}
Let $V_{q_{11},q_{12},q_{22},c}$ denote the bipodal graphon with small block size
$c$ and edge densities $q_{11},q_{12},q_{22}$ on the small block, cross edges,
and large block, respectively.  Fix $q_{11}=1/2$ and set $q_{12}=\sc(r)$.
We determine $c$ and $q_{22}$ from the two constraints by defining
\[
  F(r,\delta,c,q_{22})
  :=\bigl(e(V_{q_{11},q_{12},q_{22},c})-(r-\delta),\,
  t(H,V_{q_{11},q_{12},q_{22},c})-r^m\bigr).
\]
We will solve $F(r,\delta,c,q_{22})=(0,0)$ for $(c,q_{22})$ in terms of $(r,\delta)$
using the analytic implicit function theorem.
The density formulas are polynomial in $q_{11},q_{12},q_{22},c$, so they
extend $F$ to real $c$ near $0$.  Since $q_{12}=\sc(r)$ is analytic near
$r_0$, this extension is analytic and satisfies $F(r,0,0,r)=(0,0)$.

Writing $n=|V(H)|=2m/d$, we obtain the following formulas, with all $H$-density terms
corresponding to assignments of at least two vertices to the small block
absorbed into the $O_H(c^2)$ remainder:
\begin{align*}
  e(V_{q_{11},q_{12},q_{22},c})
  &=c^2q_{11}+2c(1-c)q_{12}+(1-c)^2q_{22},\\
  t(H,V_{q_{11},q_{12},q_{22},c})
  &=(1-c)^nq_{22}^m
  +nc(1-c)^{n-1}q_{12}^dq_{22}^{m-d}+O_H(c^2).
\end{align*}
Thus, the Jacobian of $F$ with respect to $(c,q_{22})$ at $(r,0,0,r)$ satisfies
\begin{gather*}
  \partial_{(c,q_{22})}F(r,0,0,r)=
  \begin{pmatrix}
    2(q_{12}-r) & 1 \\
    \frac{2m}{d}r^{m-d}(q_{12}^d-r^d) & m r^{m-1}
  \end{pmatrix},\\
  \det\partial_{(c,q_{22})}F(r,0,0,r)=-2m r^{m-1}D_d(r,q_{12}),
\end{gather*}
where
\[
  D_d(r,q_{12}):=\frac{q_{12}^d-r^d}{d r^{d-1}}-(q_{12}-r).
\]
Since $q_{12}=\sc(r)\ne r$ by (M2) of \Cref{thm:lz-boundary},
strict convexity of $z\mapsto z^d$ gives $D_d(r,q_{12})>0$.
In particular, the Jacobian is invertible at $r=r_0$, so the analytic implicit
function theorem at $(r_0,0,0,r_0)$ gives jointly analytic functions
$c(r,\delta)$ and $q_{22}(r,\delta)$ near $(r_0,0)$.
Local uniqueness and the identity $F(r,0,0,r)=(0,0)$ imply
$c(r,0)=0$, $q_{22}(r,0)=r$, and
\begin{equation}\label{eq:comparison-graphon-constraints}
  e(V_{q_{11},q_{12},q_{22}(r,\delta),c(r,\delta)})=r-\delta,
  \qquad
  t(H,V_{q_{11},q_{12},q_{22}(r,\delta),c(r,\delta)})=r^m.
\end{equation}
Shrink $I$ and decrease $\delta_0$ so that
$\bar I\times[-\delta_0,\delta_0]$ lies in this analytic domain.
Differentiating the constraints at $\delta=0$ and using the Jacobian gives,
with uniform Taylor remainders,
\begin{equation}\label{eq:expansion-cq}
  c(r,\delta)=\frac{\delta}{2D_d(r,q_{12})}+O_{H,I}(\delta^2),
  \qquad
  q_{22}(r,\delta)=r-\frac{2(q_{12}^d-r^d)}{d r^{d-1}}\,c(r,\delta)+O_{H,I}(c(r,\delta)^2).
\end{equation}
Since $D_d(r,\sc(r))$ is bounded above and away from zero on $\bar I$,
we may decrease $\delta_0$ so that $0<c(r,\delta)<1/2$ and $0<q_{22}(r,\delta)<1$
for every $r\in I$ and $0<\delta<\delta_0$.
Hence $W_{r,\delta}:=V_{q_{11},q_{12},q_{22}(r,\delta),c(r,\delta)}$ is a graphon
with the required density constraints.

It remains to prove \eqref{eq:quadratic-upper-bound}.
Abbreviate $c=c(r,\delta)$ and $q_{22}=q_{22}(r,\delta)$.  Since $q_{22}-r=O_{H,I}(c)$ and
all edge densities and $\pc(r)$ stay away from $0$ and $1$, the block decomposition and Taylor's theorem give
\begin{align*}
  I_{\pc(r)}(W_{r,\delta})-J_{\pc(r)}(r)
  &=c^2J_{\pc(r)}(q_{11})+2c(1-c)J_{\pc(r)}(q_{12})
    +(1-c)^2J_{\pc(r)}(q_{22})-J_{\pc(r)}(r)\\
  &=J_{\pc(r)}'(r)(q_{22}-r)
    +2c\{J_{\pc(r)}(q_{12})-J_{\pc(r)}(r)\}+O_{H,I}(c^2)\\
  &=2c\left\{J_{\pc(r)}(q_{12})-J_{\pc(r)}(r)
    -\frac{J_{\pc(r)}'(r)}{d r^{d-1}}(q_{12}^d-r^d)\right\}
    +O_{H,I}(c^2).
\end{align*}
The second equality uses Taylor expansion of $J_{\pc(r)}$ at $r$, and the last
uses the second expansion in \eqref{eq:expansion-cq}.
The bracket vanishes since $\ell_r$ touches the graph of $\varphi_{\pc(r),d}$ at $r^d$ and
$q_{12}^d$ by (M2) of \Cref{thm:lz-boundary}.
The first expansion in \eqref{eq:expansion-cq} then gives a cost difference
of $O_{H,I}(c^2)=O_{H,I}(\delta^2)$.
\end{proof}

Recall that the boundary excess $G_r$ in \eqref{eq:boundary-excess} is defined
only for $\delta>0$, since the Kenyon--Radin--Ren--Sadun regime in
\Cref{thm:krrs-analytic-extension} requires $\tau-\eps^m>0$.
In addition to the quadratic expansion, the following theorem gives a
real-analytic \emph{extension} of $G_r$ to a two-sided neighborhood of
$\delta=0$.  The $\delta$-derivatives of this extension are
used later in \Cref{sec:nonexceptional-proof}.

\begin{theorem}[Quadratic expansion of the boundary excess]\label{thm:boundary-excess-expansion}
The function $(r,\delta)\mapsto G_r(\delta)$ extends to a real-analytic
function $G:\mathcal N\to\RR$ on an open neighborhood $\mathcal N$ of
$\bar I\times[-\delta_0,\delta_0]$.  Uniformly for $r\in I$,
\begin{equation}\label{eq:boundary-excess-expansion}
  G(r,\delta)=\frac12A_H(r)\delta^2+O_{H,I}(|\delta|^3)
  \qquad(|\delta|<\delta_0),
\end{equation}
where $A_H(r):=\partial_\delta^2G(r,0)$ is real-analytic and
$\inf_{r\in I}A_H(r)>0$.
\end{theorem}

\begin{proof}
Apply \Cref{thm:krrs-analytic-extension} with $\eps_0=r_0$, as in the proof
of \Cref{lem:fixed-density-bipodality}.  It gives an open neighborhood $U$
of $r_0$, a constant $\Delta>0$, and a real-analytic parameter map
\[
  (\eps,\vartheta)\longmapsto(q_{11},q_{12},q_{22},c)
  \qquad (\eps,\vartheta)\in U\times(-\Delta,\Delta).
\]
For $\vartheta>0$, these four parameters describe the entropy maximizer,
unique up to relabeling, with edge density $\eps$ and $H$-density
$\eps^m+\vartheta$.  At $\vartheta=0$, the three edge-density parameters
lie in $(0,1)$, with $q_{22}=\eps$ and $c=0$.

To express the parameter map in terms of $(r,\delta)$, substitute
$(\eps,\vartheta)=(r-\delta,r^m-(r-\delta)^m)$.
The substitution takes values in $U\times(-\Delta,\Delta)$ on
\[
  \mathcal N_0:=\bigl\{(r,\delta)\in U\times\RR:
  r-\delta\in U,\ \lvert r^m-(r-\delta)^m\rvert<\Delta\bigr\},
\]
which is an open neighborhood of $(r_0,0)$ by continuity.
We retain the notation $q_{11},q_{12},q_{22},c$ for the resulting jointly
real-analytic functions of $(r,\delta)$ on $\mathcal N_0$.

By continuity, choose an open neighborhood $\mathcal N\subseteq\mathcal N_0$
of $(r_0,0)$ on which all $q_{ij}$ remain in $(0,1)$.
Shrink $I$ and decrease $\delta_0$ so that
$\bar I\times[-\delta_0,\delta_0]\subseteq\mathcal N$ and
apply \Cref{lem:quadratic-lower-bound,lem:quadratic-upper-bound}
for $r\in I$ and $0<\delta<\delta_0$.  Define
\[
  G(r,\delta):=
  c^2J_{\pc(r)}(q_{11})+2c(1-c)J_{\pc(r)}(q_{12})
  +(1-c)^2J_{\pc(r)}(q_{22})-J_{\pc(r)}(r)
  \qquad ((r,\delta)\in\mathcal N),
\]
where all block parameters are evaluated at $(r,\delta)$.
This function is real-analytic: $\pc$ is real-analytic on $U$ with values
in $(0,1)$, and $J_p(z)$ is real-analytic for $p,z\in(0,1)$.

For $r\in I$ and $0<\delta<\delta_0$, the substituted excess
$r^m-(r-\delta)^m$ is positive, so the parameters describe the fixed-density
bipodal optimizer $B_{r-\delta,r^m}$.
Thus by \eqref{eq:reduced-objective-attainment},
\[
  G(r,\delta)
  =I_{\pc(r)}(B_{r-\delta,r^m})-J_{\pc(r)}(r)
  =I_{\pc(r),r}(r-\delta)-J_{\pc(r)}(r)
  =G_r(\delta)
  \qquad (r\in I,\ 0<\delta<\delta_0).
\]
proving that $G$ extends $G_r$.
At $\delta=0$, we have $c(r,0)=0$ and $q_{22}(r,0)=r$, so $G(r,0)=0$.

Applying \Cref{lem:quadratic-lower-bound} to $B_{r-\delta,r^m}$ and
comparing its cost with the graphon in \Cref{lem:quadratic-upper-bound} gives
\[
  C_{d,I}\delta^2\le G(r,\delta)\le C_{H,I}\delta^2
  \qquad(r\in I,\ 0<\delta<\delta_0).
\]
Since $G$ is analytic and $G(r,0)=0$, these bounds imply
$\partial_\delta G(r,0)=0$ and
\[
  A_H(r)=\partial_\delta^2G(r,0)
  =\lim_{\delta\downarrow0}\frac{2G(r,\delta)}{\delta^2}\ge2C_{d,I}>0.
\]
Thus $A_H$ is uniformly positive on $I$ and real-analytic as a derivative of $G$.
Finally, $\partial_\delta^3G$ is bounded on the compact rectangle
$\bar I\times[-\delta_0,\delta_0]$.  Taylor's theorem at $\delta=0$ therefore
gives \eqref{eq:boundary-excess-expansion}, uniformly for $r\in I$.
\end{proof}

The function $A_H$ does not depend on the chosen neighborhood or analytic
extension: any two extensions agree for small positive $\delta$, where
both coincide with \eqref{eq:boundary-excess}, and hence their second derivatives
at $\delta=0$ agree.  Since $r_0$ was arbitrary, these local functions
define a positive real-analytic function
\begin{equation*}
  A_H:(0,1)\setminus\{r_*\}\longrightarrow(0,\infty).
\end{equation*}
